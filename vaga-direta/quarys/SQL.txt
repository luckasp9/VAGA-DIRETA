Tabela em que os estágios são inseridos após o Get ser executado

CREATE TABLE tcc.vaga (
    id SERIAL PRIMARY KEY,
    job_id VARCHAR UNIQUE,
    job_title VARCHAR(255),
    employer_name VARCHAR(255),
    job_description TEXT,
    job_city VARCHAR(100),
    job_state VARCHAR(100),
    job_country VARCHAR(100),
    job_employment_type VARCHAR(100),
    job_apply_link TEXT,
    job_salary VARCHAR(255),
    job_benefits TEXT,
    created_at TIMESTAMP DEFAULT NOW()
);

--------------------------------------------------------------------------------------------------------------------------------------------------------------------------------

Tabela de cursos

CREATE TABLE IF NOT EXISTS tcc.curso (
    id SERIAL PRIMARY KEY,
    nome TEXT UNIQUE NOT NULL
);

--------------------------------------------------------------------------------------------------------------------------------------------------------------------------------


Tabela de relação das vagas com os cursos

CREATE TABLE IF NOT EXISTS tcc.vaga_curso (
    vaga_id INTEGER REFERENCES tcc.vaga(id),
    curso_id INTEGER REFERENCES tcc.cursos(id),
    PRIMARY KEY (vaga_id, curso_id)
)
