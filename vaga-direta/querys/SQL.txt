Tabela em que os estágios são inseridos após o Get ser executado

CREATE TABLE tcc.vaga (
    id SERIAL PRIMARY KEY,
    id_vaga VARCHAR UNIQUE,
    titulo_vaga VARCHAR(255),
    empresa_nome VARCHAR(255),
    descricao_vaga TEXT,
    cidade_vaga VARCHAR(100),
    estado_vaga VARCHAR(100),
    salario VARCHAR(255),
    url VARCHAR(255),
    curso_id INTEGER,
    pcd BOOLEAN,
    modalidade VARCHAR(255),
    id_plataforma INTEGER,
    created_at TIMESTAMP DEFAULT NOW()
);

--------------------------------------------------------------------------------------------------------------------------------------------------------------------------------

Tabela de beneficios

CREATE TABLE tcc.beneficio (
    id SERIAL PRIMARY KEY,
    vaga_id INT REFERENCES tcc.vaga(id) ON DELETE CASCADE,
    beneficio TEXT NOT NULL
);
--------------------------------------------------------------------------------------------------------------------------------------------------------------------------------

Tabela de usuários

CREATE TABLE tcc.usuario (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nome VARCHAR NOT NULL,
    email VARCHAR NOT NULL,
    tipo_usuario VARCHAR NOT NULL,
    telefone VARCHAR,
    curso VARCHAR NOT NULL,
    semestre SMALLINT NOT NULL,
    estado VARCHAR NOT NULL,
    senha VARCHAR NOT NULL
);

--------------------------------------------------------------------------------------------------------------------------------------------------------------------------------

Tabela de cursos

CREATE TABLE tcc.curso (
    id SERIAL PRIMARY KEY,
    vaga_id INT REFERENCES tcc.vaga(id) ON DELETE CASCADE,
    curso TEXT 
);

--------------------------------------------------------------------------------------------------------------------------------------------------------------------------------

CREATE TABLE tcc.plataforma (
	id int4 DEFAULT nextval('tcc.curso_id_seq'::regclass) NOT NULL,
	plataforma text NOT NULL,
	CONSTRAINT curso_pkey PRIMARY KEY (id)
);
